# FA3-CDNA3 Build Instructions
# ============================================================
# Include this fragment in the ROCm build system to compile the
# FA3-CDNA3 prefill kernel with the required performance flags.
#
# Usage: include(fa3_cdna3_prefill_cmake.txt) in the appropriate
# CMakeLists.txt, or copy the relevant sections.
# ============================================================

# ---- Required compiler flags for FA3-CDNA3 kernel ----
#
# These flags are CRITICAL. Without them, the LLVM backend can silently
# destroy the hand-tuned instruction schedule.

set(FA3_CDNA3_COMPILE_FLAGS
    # C++ standard (required for CK templates and structured bindings)
    -std=c++20

    # Fast exp2 approximation: replaces 1/4-rate v_exp_f32 with a full-rate
    # polynomial approximation in the softmax hot path (~4x faster).
    -DCK_TILE_FMHA_FWD_FAST_EXP2=1

    # Flush denormals to zero in FP32 accumulator to avoid 4+ cycle penalties
    # when S_acc or O_acc produces denormal values during rescaling.
    -fgpu-flush-denormals-to-zero

    # CRITICAL: Disable the post-MI scheduler.
    # The post-MI scheduler runs AFTER instruction selection and can reorder
    # instructions across our sched_barrier() fences, collapsing the
    # compute/memory cluster structure we hand-crafted.
    -mllvm -enable-post-misched=0

    # Loop strength reduction: prevent LSR from introducing extra address
    # computation instructions that compete with MFMA for VALU slots.
    -mllvm --lsr-drop-solution=1

    # Prevent hoisting of scalar (uniform) computations that could create
    # instruction dependencies bridging cluster boundaries.
    -fno-offload-uniform-block

    # Aggressive inlining: ensures asm_primitives.hpp wrappers are truly
    # inlined and not emitted as device function calls inside the inner loop.
    -mllvm -amdgpu-early-inline-all=true

    # Eliminate device-side function call overhead (no call/return instructions
    # in the kernel inner loop for any of the asm wrapper functions).
    -mllvm -amdgpu-function-calls=false

    # Deep template nesting required for CK tile type templates.
    -fbracket-depth=1024

    # Target MI300X specifically.
    --offload-arch=gfx942
)

# ---- Source file ----
set(FA3_CDNA3_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/fa3_cdna3_prefill.cu
)

# ---- Object library ----
add_library(fa3_cdna3_kernel OBJECT ${FA3_CDNA3_SOURCES})

target_include_directories(fa3_cdna3_kernel PRIVATE
    ${CMAKE_SOURCE_DIR}/include            # flashinfer/attention/cdna3/
    ${CMAKE_SOURCE_DIR}/include/gpu_iface  # existing HIP backend headers
)

target_compile_options(fa3_cdna3_kernel PRIVATE
    ${FA3_CDNA3_COMPILE_FLAGS}
)
